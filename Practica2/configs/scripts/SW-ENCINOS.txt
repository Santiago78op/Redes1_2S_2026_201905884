! ==========================================================
! SW-ENCINOS  -  Principal Zona 5 (Encinos de Cayala) / Root VLAN 54   (carne 201905884)
! Modelo: Cisco 2960-24TT
! SCRIPT DE CONFIGURACION - listo para pegar en la CLI de Packet Tracer.
! ==========================================================
enable
configure terminal
 hostname SW-ENCINOS
 no ip domain-lookup
!
! ---- Spanning Tree: Rapid PVST+ ----
 spanning-tree mode rapid-pvst
! raiz primaria de la VLAN de su zona
 spanning-tree vlan 54 priority 24576
!
! ---- VTP: server del dominio 201905884 ----
! transparent primero: deja la revision de configuracion en 0
 vtp mode transparent
 vtp domain 201905884
 vtp password cayala2S2026
 vtp version 2
 vtp mode server
!
! ---- Troncales 802.1Q: nativa 99 y solo las VLANs necesarias ----
 interface GigabitEthernet0/1
  description -> SW-EMPRESARIAL (enlace unico)
  switchport mode trunk
  switchport trunk native vlan 99
  switchport trunk allowed vlan 54,99
  switchport nonegotiate
  no shutdown
 exit
!
! ---- VLANs que este servidor VTP aporta al dominio ----
 vlan 54
  name ENCINOS
 exit
!
! ---- Puertos de acceso VLAN 54 (ENCINOS) ----
 interface range FastEthernet0/1-7
  description Garitas y administracion del condominio
  switchport mode access
  switchport access vlan 54
  spanning-tree portfast
  spanning-tree bpduguard enable
  no shutdown
 exit
! ---- Puertos sin uso: VLAN Blackhole 999 y apagados ----
 interface range FastEthernet0/8-24
  switchport mode access
  switchport access vlan 999
  shutdown
 exit
 interface GigabitEthernet0/2
  switchport mode access
  switchport access vlan 999
  shutdown
 exit
!
! ---- Administracion del switch sobre la VLAN 99 ----
 interface Vlan99
  ip address 192.168.4.197 255.255.255.240
  no shutdown
 exit
 ip default-gateway 192.168.4.193
end
write memory

enable
configure terminal
 hostname SW-MODA-A1
 no ip domain-lookup
!
! ---- Spanning Tree: Rapid PVST+ ----
 spanning-tree mode rapid-pvst
!
! ---- VTP: client del dominio 201905884 ----
! transparent primero: deja la revision de configuracion en 0
 vtp mode transparent
 vtp domain 201905884
 vtp password cayala2S2026
 vtp version 2
 vtp mode client
!
! ---- Troncales 802.1Q: nativa 99 y solo las VLANs necesarias ----
 interface GigabitEthernet0/1
  description -> SW-MODA (uplink)
  switchport mode trunk
  switchport trunk native vlan 99
  switchport trunk allowed vlan 44,99
  switchport nonegotiate
  no shutdown
 exit
!
! ---- Puertos de acceso VLAN 44 (MODA) ----
 interface range FastEthernet0/1-24
  description Restaurantes y hotel
  switchport mode access
  switchport access vlan 44
  spanning-tree portfast
  spanning-tree bpduguard enable
  no shutdown
 exit
! ---- Puertos sin uso: VLAN Blackhole 999 y apagados ----
 interface GigabitEthernet0/2
  switchport mode access
  switchport access vlan 999
  shutdown
 exit
!
! ---- Administracion del switch sobre la VLAN 99 ----
 interface Vlan99
  ip address 192.168.4.201 255.255.255.240
  no shutdown
 exit

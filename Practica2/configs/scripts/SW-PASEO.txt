! ==========================================================
! SW-PASEO  -  Principal Zona 1 (Paseo Cayala) / Root VLAN 14   (carne 201905884)
! Modelo: Cisco 3560-24PS
! SCRIPT DE CONFIGURACION - listo para pegar en la CLI de Packet Tracer.
! ==========================================================
enable
configure terminal
 hostname SW-PASEO
 no ip domain-lookup
!
! ---- Spanning Tree: Rapid PVST+ ----
 spanning-tree mode rapid-pvst
! raiz primaria de la VLAN de su zona
 spanning-tree vlan 14 priority 24576
!
! ---- VTP: server del dominio 201905884 ----
! transparent primero: deja la revision de configuracion en 0
 vtp mode transparent
 vtp domain 201905884
 vtp password cayala2S2026
 vtp version 2
 vtp mode server
!
! ---- Po1: EtherChannel LACP 2x1G hacia SW-EMPRESARIAL ----
! El Port-channel se crea ANTES de sumar los miembros: si un miembro entra
! con una configuracion distinta a la del canal, queda suspendido (s).
 interface Port-channel1
  description Po1 -> SW-EMPRESARIAL (2x1G LACP)
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk native vlan 99
  switchport trunk allowed vlan 14,34,44,99
  switchport nonegotiate
 exit
 interface range GigabitEthernet0/1-2
  description Miembro de Po1 -> SW-EMPRESARIAL
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk native vlan 99
  switchport trunk allowed vlan 14,34,44,99
  switchport nonegotiate
  channel-protocol lacp
  channel-group 1 mode active
  no shutdown
 exit
!
! ---- Troncales 802.1Q: nativa 99 y solo las VLANs necesarias ----
 interface FastEthernet0/1
  description -> SW-PASEO-A1
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk native vlan 99
  switchport trunk allowed vlan 14,99
  switchport nonegotiate
  no shutdown
 exit
 interface FastEthernet0/2
  description -> SW-PASEO-A2
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk native vlan 99
  switchport trunk allowed vlan 14,99
  switchport nonegotiate
  no shutdown
 exit
 interface FastEthernet0/23
  description -> SW-MODA (cierra la malla del nucleo)
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk native vlan 99
  switchport trunk allowed vlan 14,44,99
  switchport nonegotiate
  no shutdown
 exit
 interface FastEthernet0/24
  description -> SW-DECIMO (enlace 2 de 2 de la zona 3)
  switchport trunk encapsulation dot1q
  switchport mode trunk
  switchport trunk native vlan 99
  switchport trunk allowed vlan 34,99
  switchport nonegotiate
  no shutdown
 exit
!
! ---- VLANs que este servidor VTP aporta al dominio ----
 vlan 14
  name PASEO
 exit
!
! ---- Puertos de acceso VLAN 14 (PASEO) ----
 interface range FastEthernet0/3-14
  description Locales del Paseo
  switchport mode access
  switchport access vlan 14
  spanning-tree portfast
  spanning-tree bpduguard enable
  no shutdown
 exit
! ---- Puertos sin uso: VLAN Blackhole 999 y apagados ----
 interface range FastEthernet0/15-22
  switchport mode access
  switchport access vlan 999
  shutdown
 exit
!
! ---- Administracion del switch sobre la VLAN 99 ----
 interface Vlan99
  ip address 192.168.4.194 255.255.255.240
  no shutdown
 exit
 ip default-gateway 192.168.4.193
end
write memory
